    function restoreBackup(event) {

      const reader = new FileReader();

      reader.onload = function(e) {

        try {

            saveTasksData();

            alert("バックアップデータを正常に復元しました。");

          } else {

          }

        } catch (err) {

          alert("JSONの読み込みに失敗しました。");

        }

      };

      reader.readAsText(file);

    }

    function generateDateArray(startStr, endStr) {

      const dates = [];

      let curr = new Date(startStr);

      const end = new Date(endStr);

      while (curr <= end) {

        dates.push(curr.toISOString().split("T")[0]);

        curr.setDate(curr.getDate() + 1);

      }

      return dates;

    }


    function checkIsDelayed(task) {

      if (task.status === "完了") return false;

      const today = new Date().toISOString().split("T")[0];

      return task.end < today;

    }


    function getStatusColorClass(status) {

      switch (status) {

        case "完了": return "bg-emerald-500";

        case "進行中": return "bg-blue-500";

        case "保留中": return "bg-amber-500";

        default: return "bg-slate-400";

      }

    }


    function escapeHTML(str) {

      return str.replace(/[&<>"']/g, function(m) {

        return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m];

      });

    }
